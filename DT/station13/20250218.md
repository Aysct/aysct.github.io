### solution

显然，题目给出了 $F=\prod_{i}(1+a_ix)$。

根据常见的经验，我们有时会用 exp-ln 方法将乘积式转化成和式来加速运算。

同理，我们有一个经典的 Trick：通过 ln/exp-ln 将式子转为和式来讨论式子的性质。

于是：
$$
\ln{F}=\sum_i \ln(1+a_ix)\\
\ln{F}=\sum_i \sum_{j \geq 1}\frac{(-1)^{j+1}a_i^jx^j}{j}\\
\ln{F}=\sum_{j \geq 1} \frac{(-1)^{j+1}x^j}{j}\sum_i a_i^j\\
$$
我们发现题目求的就是 $\frac{m}{(-1)^{m+1}}[x^m]\ln{F}$。

但注意到 $m \leq 10^9$，我们不能直接套多项式 ln，而是需要找一个快速求单点的方法。

考察多项式 ln 的过程，我们知道：
$$
(\ln{F})'=\frac{F'}{F}\\
[x^m]\ln{F}=\frac{1}{m}[x^{m-1}]\frac{F'}{F}
$$
 通过 bostan-mori，我们很容易在 $O(n\log{n}\log{m})$ 的时间求出 $[x^{m-1}]\frac{F'}{F}$，于是：
$$
ans=\frac{1}{(-1)^{m+1}}[x^{m-1}]\frac{F'}{F}=(-1)^{m+1}[x^{m-1}]\frac{F'}{F}
$$
就好做了。

### 实现

都是一些比较常见的东西，比较熟悉的话一遍过不难。

弱化版把 `mul` 改成暴力实现就好了。

```cpp
#include<bits/stdc++.h>
const int mod=899678209,g=7;
int fpow(int a,int b=mod-2){
    int r=1;
    while(b){
        if(b&1)r=1ll*r*a%mod;
        a=1ll*a*a%mod;
        b>>=1;
    }
    return r;
}
using namespace std;
namespace Poly{
#define poly vector<int>
    const int N=131072;
    int sz,w[N+5];
    void ntt(poly &a){
        for(int len=sz,mid=sz>>1;len>=2;len>>=1,mid>>=1){
            int wn=fpow(g,(mod-1)/len);
            for(int i=w[0]=1;i<mid;i++)w[i]=1ll*w[i-1]*wn%mod;
            for(int i=0;i<sz;i+=len){
                for(int j=0;j<mid;j++){
                    int x=a[i+j],y=a[i+j+mid];
                    a[i+j]=(x+y)%mod;
                    a[i+j+mid]=1ll*(x-y+mod)%mod*w[j]%mod;
                }
            }
        }
    }
    void intt(poly &a){
        for(int len=2,mid=1;len<=sz;len<<=1,mid<<=1){
            int wn=fpow(fpow(g,(mod-1)/len));
            for(int i=w[0]=1;i<mid;i++)w[i]=1ll*w[i-1]*wn%mod;
            for(int i=0;i<sz;i+=len){
                for(int j=0;j<mid;j++){
                    int x=a[i+j],y=1ll*w[j]*a[i+j+mid]%mod;
                    a[i+j]=(x+y)%mod;
                    a[i+j+mid]=(x-y+mod)%mod;
                }
            }
        }
        int t=fpow(sz);
        for(int i=0;i<sz;i++)a[i]=1ll*a[i]*t%mod;
    }
    poly mul(poly a,poly b){
        int n=(int)a.size()+(int)b.size()-1;
        sz=1;while(sz<n)sz<<=1;a.resize(sz),b.resize(sz);
        ntt(a),ntt(b);for(int i=0;i<sz;i++)a[i]=1ll*a[i]*b[i]%mod;intt(a);
        a.resize(n);
        return a;
    }
    int mori(poly f,poly g,int n){
        for(;n;n>>=1){
            if(f.size()>=n+1)f.resize(n+1);
            if(g.size()>=n+1)g.resize(n+1);
            poly h=g;
            for(int i=1;i<(int)h.size();i+=2)h[i]=(mod-h[i])%mod;
            f=mul(f,h);g=mul(g,h);
            int i;
            for(i=n&1;i<(int)f.size();i+=2)f[i>>1]=f[i];
            f.resize(i>>1);
            for(i=0;i<(int)g.size();i+=2)g[i>>1]=g[i];
            g.resize(i>>1);
        }
        if(f.empty())return 0;
        return 1ll*f[0]*fpow(g[0])%mod;
    }
    poly dev(poly a){
        for(int i=0;i<(int)a.size()-1;i++)a[i]=1ll*(i+1)*a[i+1]%mod;
        a.pop_back();
        return a;
    }
}
using namespace Poly;
int n,m,ans;
poly a,b;
signed main(){
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    cin>>n>>m;b.resize(n+1);
    for(int i=1;i<=n;i++)cin>>b[i];
    b[0]=1;
    a=dev(b);
    ans=mori(a,b,m-1);
    if((m+1)&1)ans=(mod-ans)%mod;
    cout<<ans;
    return 0;
}
```
